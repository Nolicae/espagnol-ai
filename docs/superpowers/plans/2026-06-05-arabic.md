```python
        return _ALL_BY_NAME[name]

    pool = EDGE_VOICES if is_online else []

    if language_code:
        lc_lower = language_code.lower()
        candidates = [v for v in pool if any(lc.lower().startswith(lc_lower) for lc in v["languageCodes"])]

        candidates = pool

    if ssml_gender and candidates:
        gendered = [v for v in candidates if v["ssmlGender"] == ssml_gender]
        if gendered:
```

```javascript
// Vercel serverless — arabe-tts HF Space proxy
const TTS_BASE_URL = 'https://nolicae-arabe-tts.hf.space';
const DEFAULT_VOICE = 'ar-SA-ZariyahNeural';

async function arabeTTS(text, voice = DEFAULT_VOICE) {
  const response = await fetch(`${TTS_BASE_URL}/v1/text:synthesize`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      input: { text },
      voice: { name: voice },
      audioConfig: { audioEncoding: 'MP3' }
    })
  });
  if (!response.ok) throw new Error(`arabe-tts ${response.status}`);
  const data = await response.json();
  return Buffer.from(data.audioContent, 'base64');
}

module.exports = async function handler(req, res) {
  const { text, voice = DEFAULT_VOICE } = req.query;
  if (!text) return res.status(400).end('missing text');
  try {
    const audio = await arabeTTS(text, voice);
    res.setHeader('Content-Type', 'audio/mpeg');
    res.setHeader('Cache-Control', 'no-store');
    res.end(audio);
  } catch (e) {
    console.error('TTS-AR error:', e.message);
    res.status(500).end('TTS error');
  }
};
```
